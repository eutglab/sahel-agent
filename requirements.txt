# SAHEL Agent — runtime dependencies
# Every dependency has a reason. No heavy infra.
# Deliberately >= (not pinned): deploy targets range from local Python 3.10 to
# Streamlit Community Cloud's Python 3.14, and numpy/pandas need the freedom
# to resolve a version with a prebuilt wheel for whichever is in use — a
# strict pin to an older release with no cp314 wheel forces a source build
# that can hang for many minutes on Cloud. Verified 2026-09-12 on both.

pyyaml>=6.0            # configurable sensor thresholds + defaults
pillow>=10.0           # local image heuristic (no ML model)
numpy>=1.24            # pixel math for the local vision heuristic
pandas>=2.0            # benchmark tables + run history views
requests>=2.31         # Open-Meteo / Exa (keyless-capable) HTTP calls
